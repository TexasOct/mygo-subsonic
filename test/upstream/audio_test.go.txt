// Original bytes pass from MoeJS URLs through the official decoder, including seeking.
func TestSubsonicOriginalAudio(t *testing.T) {
	for _, format := range []struct{ suffix, mime, quality string }{
		{"mp3", "audio/mpeg", "high"},
		{"flac", "audio/flac", "lossless"},
		{"ogg", "audio/ogg; codecs=vorbis", "high"},
	} {
		t.Run(format.suffix, func(t *testing.T) {
			data, err := os.ReadFile("testdata/seek." + format.suffix)
			if err != nil {
				t.Fatal(err)
			}
			server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				if err := r.ParseForm(); err != nil {
					t.Error(err)
				}
				sum := md5.Sum([]byte("password" + r.Form.Get("s")))
				if r.Form.Get("u") != "listener" || r.Form.Get("s") == "" || r.Form.Get("t") != hex.EncodeToString(sum[:]) {
					t.Error("missing audio or metadata authentication")
				}
				method := strings.TrimPrefix(r.URL.Path, "/rest/")
				if method == "stream.view" {
					if r.Method != "GET" || r.Form.Get("id") != "s /&+" || r.Form.Get("format") != "raw" || r.Form.Has("maxBitRate") || r.Form.Has("estimateContentLength") {
						t.Error("original audio request was changed")
					}
					w.Header().Set("Content-Type", format.mime)
					w.Write(data)
					return
				}
				if r.Method != "POST" {
					t.Error("metadata must use POST")
				}
				response := map[string]any{"status": "ok", "version": "1.16.1"}
				switch method {
				case "ping.view":
				case "getSong.view":
					if r.Form.Get("id") != "s /&+" {
						t.Error("track ID was changed")
					}
					response["song"] = map[string]any{"id": "s /&+", "suffix": format.suffix, "contentType": format.mime, "size": len(data), "duration": 5}
				default:
					t.Errorf("unexpected method: %s", method)
				}
				w.Header().Set("Content-Type", "application/json")
				json.NewEncoder(w).Encode(map[string]any{"subsonic-response": response})
			}))
			defer server.Close()
			p, err := plugin.Load(os.Getenv("MYGO_SUBSONIC_DIR"), t.TempDir())
			if err != nil {
				t.Fatal(err)
			}
			defer p.Close()
			for key, value := range map[string]string{"serverUrl": server.URL, "username": "listener", "password": "password"} {
				if err := p.SetSetting(key, value); err != nil {
					t.Fatal(err)
				}
			}
			if _, err := p.SignIn(); err != nil {
				t.Fatal(err)
			}
			ref, err := openStream(downloaded(data), 5000)
			if err != nil {
				t.Fatal(err)
			}
			defer ref.Close()
			original := subsonicPCM(t, ref)
			for _, quality := range []string{"standard", "high", "lossless"} {
				t.Run(quality, func(t *testing.T) {
					st, err := p.Stream("s /&+", quality)
					if err != nil {
						t.Fatal(err)
					}
					u, err := url.Parse(st.URL)
					if err != nil || u.Query().Get("format") != "raw" || st.Type != format.suffix || st.Quality != format.quality || st.Size != int64(len(data)) {
						t.Fatalf("incorrect original stream: %+v %v", st, err)
					}
					dl := (*Cache)(nil).download("subsonic", "s /&+", quality, st)
					s, err := openStream(dl, 5000)
					if err != nil {
						dl.Close()
						t.Fatal(err)
					}
					defer s.Close()
					got := subsonicPCM(t, s)
					if len(got) != len(original) {
						t.Fatalf("decoded %d samples, want %d", len(got), len(original))
					}
					for i := range got {
						if got[i] != original[i] {
							t.Fatalf("original sample changed at %d", i)
						}
					}
					if err := s.SetPosition(int64(s.SampleRate())); err != nil {
						t.Fatal(err)
					}
					buf := make([]float32, 4096)
					n, err := s.Read(buf)
					if err != nil || n == 0 {
						t.Fatalf("seek: %d samples, %v", n, err)
					}
				})
			}
		})
	}
}

func subsonicPCM(t *testing.T, s stream) []float32 {
	t.Helper()
	var result []float32
	buf := make([]float32, 4096)
	for {
		n, err := s.Read(buf)
		result = append(result, buf[:n]...)
		if err != nil && err != io.EOF {
			t.Fatal(err)
		}
		if err != nil || n == 0 {
			break
		}
	}
	var energy float64
	for _, sample := range result {
		if math.IsNaN(float64(sample)) || math.IsInf(float64(sample), 0) {
			t.Fatal("invalid PCM sample")
		}
		energy += math.Abs(float64(sample))
	}
	if len(result) < s.SampleRate()*2 || energy == 0 {
		t.Fatalf("empty or silent PCM: %d samples", len(result))
	}
	return result
}
